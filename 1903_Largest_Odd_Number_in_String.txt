class Solution {
    public String largestOddNumber(String num) {
        int n=num.length();
        // TC-O(N) SC=O(1)
        for(int i=n-1;i>=0;i--){
            int c=num.charAt(i);
            if((c-'0')%2!=0){
                return num.substring(0,i+1);
            }
        }
       return "";
    }
}
